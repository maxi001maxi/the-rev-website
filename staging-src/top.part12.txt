<script>(() => {
 const reduce=matchMedia('(prefers-reduced-motion:reduce)');
 const seen=new WeakSet();
 let observer=null,booted=false,header=null,pending=false;
 function headerState(){
  pending=false;
  if(header)header.classList.toggle('rev-is-scrolled',scrollY>24);
 }
 function scheduleHeader(){if(!pending){pending=true;requestAnimationFrame(headerState);}}
 function discover(){
  if(!booted)return;
  const selectors=document.querySelector('.rev-home-hero')
   ?'main h2,main article,main figure'
   :'.rv-section h2,.rv-benefit-grid article,.rv-oxygen-compare figure,.rv-equipment-photo,.rv-denba-product,.rv-place figure';
  document.querySelectorAll(selectors).forEach((el,index)=>{
   if(seen.has(el))return;
   seen.add(el);el.dataset.revReveal='';
   el.style.setProperty('--rev-reveal-delay',(index%3)*65+'ms');
   if(reduce.matches||!observer)return;
   const rect=el.getBoundingClientRect();
   // Already-visible content stays visible; animate only approaching sections.
   if(rect.height>0&&rect.top>=innerHeight-24){el.classList.add('rev-reveal-pending');observer.observe(el);}
  });
 }
 function boot(){
  const home=document.querySelector('.rev-home-hero');
  if(home&&!home.closest('.sc-host'))return;
  if(!document.querySelector('main h1'))return;
  if(!booted){
   booted=true;
   document.documentElement.classList.toggle('rev-motion-ready',!reduce.matches);
   if('IntersectionObserver' in window)observer=new IntersectionObserver(entries=>{
    for(const entry of entries)if(entry.isIntersecting){entry.target.classList.add('rev-is-revealed');observer.unobserve(entry.target);}
   },{rootMargin:'0px 0px -30px 0px',threshold:.08});
   header=document.querySelector('#site-header.rev-home-header');
   if(header){window.addEventListener('scroll',scheduleHeader,{passive:true});window.addEventListener('resize',scheduleHeader);headerState();}
  }
  discover();
 }
 const start=()=>{boot();new MutationObserver(boot).observe(document.body,{childList:true,subtree:true});};
 if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',start,{once:true});else start();
 reduce.addEventListener('change',()=>{
  document.documentElement.classList.toggle('rev-motion-ready',!reduce.matches);
  if(reduce.matches)document.querySelectorAll('.rev-reveal-pending').forEach(el=>el.classList.add('rev-is-revealed'));
 });
})();
